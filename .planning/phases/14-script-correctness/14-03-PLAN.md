import struct

def _make_box(box_type: str, size: int) -> bytes:
    """Create an MP4 box header with optional padding content."""
    header = struct.pack(">I", size) + box_type.encode("ascii")
    padding = b"\x00" * (size - 8) if size > 8 else b""
    return header + padding

def _make_extended_box(box_type: str, actual_size: int) -> bytes:
    """Create an MP4 box with 64-bit extended size (size field == 1)."""
    header = struct.pack(">I", 1) + box_type.encode("ascii") + struct.pack(">Q", actual_size)
    padding = b"\x00" * (actual_size - 16) if actual_size > 16 else b""
    return header + padding


class TestFaststart:
    def test_faststart_moov_first(self, tmp_path):
        f = tmp_path / "moov_first.mp4"
        f.write_bytes(_make_box("ftyp", 16) + _make_box("moov", 24) + _make_box("mdat", 32))
        assert has_faststart(f) is True

    def test_faststart_mdat_first(self, tmp_path):
        f = tmp_path / "mdat_first.mp4"
        f.write_bytes(_make_box("ftyp", 16) + _make_box("mdat", 32) + _make_box("moov", 24))
        assert has_faststart(f) is False

    def test_faststart_extended_size(self, tmp_path):
        f = tmp_path / "extended.mp4"
        f.write_bytes(_make_box("ftyp", 16) + _make_extended_box("free", 32) + _make_box("moov", 24) + _make_box("mdat", 32))
        assert has_faststart(f) is True

    def test_faststart_empty_file(self, tmp_path):
        f = tmp_path / "empty.mp4"
        f.write_bytes(b"")
        assert has_faststart(f) is False

    def test_faststart_malformed(self, tmp_path):
        f = tmp_path / "bad.mp4"
        f.write_bytes(b"\x00\x01\x02\x03")
        assert has_faststart(f) is False

    def test_faststart_no_moov_no_mdat(self, tmp_path):
        f = tmp_path / "ftyp_only.mp4"
        f.write_bytes(_make_box("ftyp", 16))
        assert has_faststart(f) is False

**Step 2 (GREEN): Create scripts/audit-faststart.py.**

The script provides:
1. `has_faststart(file_path: Path) -> bool` — binary parsing of MP4 box headers. Walks boxes by reading 8-byte headers (4-byte big-endian size + 4-byte ASCII type). Returns True if `moov` appears before `mdat`. Handles:
   - `size == 0`: box extends to EOF, return False
   - `size == 1`: read 8 more bytes for 64-bit extended size via `struct.unpack(">Q", ...)`
   - `size < 8` (and not 0 or 1): malformed, return False
   - IOError / struct.error: return False

2. `audit_directory(dir_path: Path) -> dict` — scans all .mp4 files in a directory tree, runs has_faststart on each, returns `{"pass": [...], "fail": [...], "error": [...]}`.

3. CLI mode with argparse:
   ```
   python3 scripts/audit-faststart.py /path/to/directory
   python3 scripts/audit-faststart.py --file /path/to/single.mp4
   ```
   Prints a table of results: filename, faststart status (PASS/FAIL), file size.
   Exits with code 0 if all pass, code 1 if any fail.

Full has_faststart implementation (from research):
```python
import struct
from pathlib import Path

def has_faststart(file_path: Path) -> bool:
    try:
        with open(file_path, "rb") as f:
            while True:
                header = f.read(8)
                if len(header) < 8:
                    return False
                size = struct.unpack(">I", header[:4])[0]
                box_type = header[4:8].decode("ascii", errors="replace")
                if box_type == "moov":
                    return True
                if box_type == "mdat":
                    return False
                if size == 0:
                    return False
                if size == 1:
                    ext = f.read(8)
                    if len(ext) < 8:
                        return False
                    size = struct.unpack(">Q", ext)[0]
                    f.seek(size - 16, 1)
                else:
                    if size < 8:
                        return False
                    f.seek(size - 8, 1)
    except (IOError, struct.error):
        return False
```
